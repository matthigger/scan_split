\ifthenelse{\equal{\pg}{q3}}{%
  \namebox
  \medskip
  \textbf{Problem 3 [10 pts] Pass or fail} \hfill (two pages)\par\smallskip
  A logistic model predicts whether a student passes from the hours $x$
  they studied:
  \[ P(\text{pass} \mid x) = \sigma(w x + b), \qquad
     \sigma(z) = \frac{1}{1 + e^{-z}}, \]
  with current parameters $w = \pick{0.8}{0.5}$ and $b = \pick{-2}{-1.5}$.
  The training set is

  \medskip
  \begin{center}
    % the versions' tables differ in shape, not only in numbers, so that
    % without templates their averages split apart
    \pick{%
      \begin{tabular}{l | c c c c}
        student & 1 & 2 & 3 & 4\\
        \hline
        hours $x_i$ & 1 & 2 & 4 & 5\\
        passed $y_i$ & 0 & 0 & 1 & 1\\
      \end{tabular}}{%
      \begin{tabular}{c c c}
        \toprule
        student & hours $x_i$ & passed $y_i$\\
        \midrule
        1 & 1 & 0\\ 2 & 2 & 0\\ 3 & 3 & 1\\ 4 & 5 & 1\\ 5 & 6 & 1\\
        \bottomrule
      \end{tabular}}
  \end{center}

  \begin{enumerate}
    \item[(a)] What does the model give for $P(\text{pass} \mid
      x = \pick{3}{4})$?
    \pick{\vspace{2.6in}}{\vspace{2.1in}}
    \item[(b)] Compute the log-loss
      $L = -\sum_i \left[ y_i \log p_i + (1 - y_i) \log(1 - p_i) \right]$,
      where $p_i = \sigma(w x_i + b)$.
  \end{enumerate}
  \vfill
  \begin{center}4\end{center}
  \newpage
  \namebox
  \medskip
  \textbf{Problem 3 (continued)}\par\smallskip
  \begin{enumerate}
    \item[(c)] The gradient is
      $\partial L / \partial w = \sum_i (p_i - y_i)\, x_i$ and
      $\partial L / \partial b = \sum_i (p_i - y_i)$. Take one gradient
      step with learning rate $\eta = \pick{0.1}{0.05}$: give the new $w$
      and $b$.
    \pick{\vspace{2.4in}}{\vspace{1.7in}}
    \item[(d)] Sketch $P(\text{pass} \mid x)$ under the
      \emph{original} $(w, b)$ on the axes below, and mark the hours at

  \end{enumerate}
